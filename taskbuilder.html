删除按钮

function btnDelete_onClick() {
    let keys = grid1.getCheckedKeys();
    let idList = Array.from(new Set(keys.split(','))).map(e => ({ id: e }));
    serviceDelete.setArg('ids', idList);
    serviceDelete.request(null, function (req, res) {
        if (res.code == 0) {
            grid1.reloadData();
            tfp.showMsg(‘删除成功’);
        }
    });
}

合并新增/更新界面

列表页操作

  1. 新增按钮点击事件中替换标准代码

  2. 修改按钮点击事件中替换标准代码(其中提示的data.sorderno 需要替换成具体的字段名)

详情页操作

  1. 另存已有的update.tfp修改页面为 edit.tfp

  2. 在对应文件夹中，添加一个名为edit_form1_Save的后台服务(必要参数id/editType 参考已有的)

  3. form1表单的数据查询属性，要设置为禁止自动加载

  4. form1表单的数据提交属性，要设置后台服务，然后指定后台服务为刚刚创建的 edit_form1_Save，请求参数必须添加(id=URL参数，editType=URL参数)

  5. page1组件的页面加载后事件中代码填入 标准代码

  6. form1表单的数据绑定后事件中代码填入 标准代码

  7. form1表单的数据提交后事件中代码填入 标准代码

  8. 单据编号处理，需要把现有的单号组件删除，拖入一个编码规则组件

    1. 组件编号与字段名匹配

    2. 数据绑定填入(字段名)

组件

tree

-- 第一部分：年份数据；第二部分：月份数据（使用ROW_NUMBER）
SELECT 
    iYear AS id,
    iYear AS name,
    0 AS parent_id 
FROM dbo.fiFinancePeriod 
GROUP BY iYear
​
UNION ALL
​
SELECT 
    sYearMonth AS id,
    sYearMonth AS name,
    iYear AS parent_id
FROM (
    SELECT DISTINCT 
        iYear, 
        sYearMonth 
    FROM dbo.fiFinancePeriod
) AS distinct_months
​
ORDER BY parent_id, id;
